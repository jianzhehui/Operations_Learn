编写备份脚本

```bash
vim /opt/backup/backup-gitlab.sh
```

```bash
#!/bin/bash
set -euo pipefail

GITLAB=gitlab
CACHE=/opt/backup/gitlab/cache
DAILY=/opt/backup/gitlab/daily
NUM=7
REPO=eladmin/eladmin

echo "backup start at $(date '+%F %T')"

if [ -d "$CACHE/$REPO.git" ]; then
	git -C "$CACHE/$REPO.git" remote update --prune
else
	mkdir -p $(dirname "$CACHE/$REPO.git")
	git clone --mirror "$GITLAB:$REPO.git" "$CACHE/$REPO.git"
fi

BACKUP="$DAILY/$(date +%F)"
rm -rf "$BACKUP"
mkdir -p "$BACKUP"
tar -czf "$BACKUP/eladmin.tar.gz" -C "$CACHE/$REPO.git" .

DEL=$(ls -t $DAILY | tail -n +$((NUM+1)))
rm -rf "$DEL"

echo "backup end at $(date '+%F %T'), now have $(ls "$DAILY" | wc -l) backup"
```

```bash
vim /opt/backup/backup-mysql.sh
```

```bash
#!/bin/bash
set -euo pipefail:

DBSRC=mysql-src
DAILY=/opt/backup/mysql/daily
NUM=30
DB=eladmin
CNF=/home/user/.my.cnf

echo "backup start at $(date '+%F %T')"

BACKUP="$DAILY/$(date +%F)"
rm -rf "$BACKUP"
mkdir -p "$BACKUP"
ssh $DBSRC "mysqldump --defaults-file=$CNF --single-transaction --no-tablespaces --databases $DB" | gzip > "$BACKUP/$DB.sql.gz"

DEL=$(ls -d | tail -n +$((NUM + 1)))
rm -rf "$DEL"

echo "backup end at $(date '+%F %T'), now have $(ls "$DAILY" | wc -l) backup"
```

配置自动执行

```bash
vim /etc/cron.d/backup
```

```bash
SHELL=/bin/bash
PATH=/usr/local/bin:/usr/local/sbin:/usr/bin:/usr/sbin:/sbin:/bin

0 2 * * * user /opt/backup/backup-gitlab.sh >> /opt/backup/log/gitlab.logs 2 &> 1
0 3 * * * user /opt/backup/backup-mysql.sh >> /opt/backup/log/mysql.logs 2 &> 1
```
